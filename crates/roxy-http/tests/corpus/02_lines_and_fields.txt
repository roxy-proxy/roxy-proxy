# Line structure, header names and values.

=== bare_lf_line_endings
expect: error bare_lf
---
GET http://example.com/ HTTP/1.1\n
Host: example.com\n
\n

=== bare_lf_after_request_line
expect: error bare_lf
---
GET http://example.com/ HTTP/1.1\r\n
Host: example.com\n
\r\n

=== bare_lf_final
expect: error bare_lf
---
GET http://example.com/ HTTP/1.1\r\n
Host: example.com\r\n
\n

=== bare_cr_in_value
expect: error bare_cr
---
GET http://example.com/ HTTP/1.1\r\n

X-A: a\rb\r\n
\r\n

=== bare_cr_line_ending
expect: error bare_cr
---

\r\n

=== crlf_injection_value
# A CRLF inside what the client thinks is one value just ends the line;
# the injected line must itself be a valid field. Here it is not.
expect: error invalid_header_name
---
GET http://example.com/ HTTP/1.1\r\n
Host: example.com\r\n
X-A: a\r\n
GET /admin HTTP/1.1\r\n
\r\n

=== values_trimmed_names_lowercased

=== hop_by_hop_stripped
expect: request GET http://example.com:80/ nohdr:connection nohdr:keep-alive nohdr:x-private nohdr:te nohdr:upgrade nohdr:proxy-authorization nohdr:proxy-connection nohdr:trailer hdr:accept=*/*
---
GET http://example.com/ HTTP/1.1\r\n
Host: example.com\r\n
Connection: keep-alive, X-Private\r\n
Keep-Alive: timeout=5\r\n
X-Private: secret\r\n
TE: trailers\r\n
Trailer: x\r\n
Upgrade: h2c\r\n
Proxy-Authorization: Basic eDp5\r\n
Proxy-Connection: keep-alive\r\n
Accept: */*\r\n
\r\n

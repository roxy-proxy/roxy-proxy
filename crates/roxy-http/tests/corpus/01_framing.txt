# Content-Length / Transfer-Encoding framing.
# Physical newlines are ignored; \r\n is written explicitly.

=== cl_te_classic
# CL.TE: front-end uses CL, back-end uses TE.
expect: error cl_and_te
---
POST http://example.com/ HTTP/1.1\r\n
Host: example.com\r\n
Content-Length: 6\r\n
Transfer-Encoding: chunked\r\n
\r\n
0\r\n
\r\n
G

=== connection_content_length_keeps_framing
# Nominating content-length as hop-by-hop drops the field from the
# forwarded head but not from the framing: the body stays a body.
expect: request POST http://example.com:80/one body=hello
expect: request GET http://example.com:80/two
expect: eof
---
POST http://example.com/one HTTP/1.1\r\n
Host: example.com\r\n
Content-Length: 5\r\n
Connection: content-length\r\n
\r\n
hello
GET http://example.com/two HTTP/1.1\r\n
Host: example.com\r\n
\r\n
